type duplexCaptureSession struct{*fakeRealtimeSession;frames atomic.Int64;bytes atomic.Int64}
func(s *duplexCaptureSession)SendAudio(pcm []byte)error{s.frames.Add(1);s.bytes.Add(int64(len(pcm)));return nil}
func TestTelephonyInternalAudioCoreFixture(t *testing.T){
 ready:=os.Getenv("TELEPHONY_INTERNAL_READY");if ready==""{t.Skip("local helper only")}
 t.Setenv("TELEMETRY_URL","");t.Setenv("TELEMETRY_LIVE_URL","");t.Setenv("SERVER_URL","")
 rt,fake:=completionFixture(t)
 input:=make(chan []byte,64);output:=make(chan RealtimeAudioFrame,64);control:=make(chan string,16)
 rt.audioIn,rt.audioOut,rt.audioControl=input,output,control
 rt.bridgeConnected=false;rt.Thinker.threadID="voice"

 rt.capabilities=RealtimeCapabilities{ContinuousConversation:true,RendererInterruption:true,CoreExecution:true}
 parent:=newTestThinker();defer parent.Stop()

 runDone:=make(chan struct{})
 var started bool;var mu sync.Mutex;var caller []byte

 mux:=http.NewServeMux()
 mux.HandleFunc("/new",func(w http.ResponseWriter,r *http.Request){
  mu.Lock();defer mu.Unlock();if started{http.Error(w,"single fixture session",409);return}
  var cfg struct{Live bool `json:"live"`};if json.NewDecoder(r.Body).Decode(&cfg)!=nil{http.Error(w,"invalid",400);return}
  if cfg.Live {
   if os.Getenv("TELEPHONY_AGENT_AUDIO_LIVE")!="1"||os.Getenv("OPENAI_API_KEY")==""{http.Error(w,"live opt-in required",403);return}
   key:=os.Getenv("OPENAI_API_KEY")
   caller=renderLiveCallerText(t,key,"Attendez, je voudrais simplement prendre rendez-vous mardi. Quel jour ai-je demandé ?")
   backend:=NewOpenAINativeProvider(key);applyModelOverrides(backend,map[string]string{"large":"gpt-6.1-sol","medium":"gpt-6.1-sol","small":"gpt-6.1-sol"})
   rt.Thinker.provider=backend;rt.pool=&ProviderPool{providers:map[string]LLMProvider{"openai":backend},default_:"openai",order:[]string{"openai"}}
   rt.Thinker.directive="Vous êtes un accueil téléphonique générique. Parlez uniquement en français. Écoutez les interruptions, puis répondez brièvement. N'inventez aucune réservation. Ne terminez pas la conversation."
   rt.messages[0].Content=rt.Thinker.directive
   rt.provider=NewOpenAIRealtimeProvider(key);rt.opts.Model,rt.opts.Voice="gpt-live-1","brise"
   rt.capabilities=capabilitiesForRealtime(rt.provider,rt.opts.Model);rt.registry=NewToolRegistry("")

   if err:=rt.openSession(false);err!=nil{http.Error(w,"live startup failed",502);return}
  }
  token:=registerAudioBridge("voice",input,output,control)
  started=true;go func(){defer close(runDone);rt.Run()}()
  _=json.NewEncoder(w).Encode(map[string]string{"thread":"voice","token":token})
 })
 mux.HandleFunc("/caller",func(w http.ResponseWriter,r *http.Request){mu.Lock();defer mu.Unlock();w.Header().Set("Content-Type","application/octet-stream");w.Write(caller)})
 mux.HandleFunc("/emit",func(w http.ResponseWriter,r *http.Request){
  var cfg struct{Item string `json:"item"`;MS int `json:"ms"`;Silent bool `json:"silent"`};if json.NewDecoder(r.Body).Decode(&cfg)!=nil||cfg.MS<=0||cfg.MS>5000{http.Error(w,"invalid",400);return}
  audio:=make([]byte,cfg.MS*48);for i:=0;i<len(audio)/2;i++{sample:=int16(10000*math.Sin(2*math.Pi*500*float64(i)/24000));if cfg.Silent{sample=0};audio[2*i]=byte(sample);audio[2*i+1]=byte(uint16(sample)>>8)}
  fake.events<-RealtimeEvent{Type:RealtimeEventAudioOut,ItemID:cfg.Item,ResponseID:cfg.Item,Audio:audio,AudioSilence:cfg.Silent};w.WriteHeader(204)
 })
 mux.HandleFunc("/interrupt",func(w http.ResponseWriter,r *http.Request){rt.rendererSpeechStarted();w.WriteHeader(204)})
 mux.HandleFunc("/stats",func(w http.ResponseWriter,r *http.Request){
  pending,awaiting,delivery:=rt.delivery.snapshot()

 if err:=os.WriteFile(ready,[]byte(srv.URL),0600);err!=nil{t.Fatal(err)}
 <-stop;unregisterAudioBridge("voice");rt.Stop();rt.cancel();rt.Thinker.Stop();if started {select{case<-runDone:case<-time.After(3*time.Second):t.Error("Core worker did not stop")}};srv.Close()
}
