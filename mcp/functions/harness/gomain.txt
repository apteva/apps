// Apteva Functions — Go worker harness.
//
// Embedded in the functions sidecar and staged as harness.go into a
// Go function's build dir, next to the user's entry.go. Both files
// are `package main`, so `go build` compiles them together: this file
// provides main(), the Context type, and the socketpair protocol; the
// user's entry.go provides `func Handle(event, ctx) (any, error)`.
//
// Protocol (4-byte big-endian length prefix + JSON, both directions):
//   sidecar -> worker:  { id, event }                       (invocation)
//                       { type:"call_result", callId, ok, result?, error? }
//   worker  -> sidecar:  { type:"ready", ok }               (once, on boot)
//                        { type:"call", callId, app, tool, input }
//                        { type:"integration", callId, conn, tool, input }
//                        { type:"stream_start", id, statusCode, headers }
//                        { type:"stream_chunk", id, encoding, data }
//                        { id, ok, result?, error?, logs? }  (invocation result)

package main

import (
	"bytes"
	"encoding/base64"
	"encoding/binary"
	"encoding/json"
	"fmt"
	"io"
	"net"
	"os"
	"runtime/debug"
	"strings"
	"sync"
)

const maxHarnessFrame = 8 << 20
const maxHarnessLogs = 16 << 10

// Context is handed to Handle. Call reaches other Apteva apps through
// the sidecar — the worker never holds a platform token itself.
type Context struct {
	FunctionName string
	FunctionID   string
	Runtime      string
	Env          map[string]string

	h             *harness
	invocationID  int64
	streamMu      sync.Mutex
	streamStarted bool
	done          chan struct{}
	logMu         sync.Mutex
	logs          []string
	logBytes      int
}

// Call invokes another Apteva app's MCP tool. input is marshalled to
// JSON; the result is the tool's (unwrapped) JSON response.
func (c *Context) Call(app, tool string, input any) (json.RawMessage, error) {
	if err := c.checkActive(); err != nil {
		return nil, err
	}
	return c.h.call(c.invocationID, app, tool, input)
}

// Integration invokes a tool on an integration connection in the
// function's project. conn may be a numeric connection id (int /
// int64) or an app slug (string, e.g. "pushover") — the sidecar
// resolves slugs to the single matching connection. Returns the
// upstream tool's response data; errors when the upstream tool
// returns non-success or the slug doesn't resolve to exactly one
// connection.
func (c *Context) Integration(conn any, tool string, input any) (json.RawMessage, error) {
	if err := c.checkActive(); err != nil {
		return nil, err
	}
	return c.h.integration(c.invocationID, conn, tool, input)
}

// Log appends a line to the captured output returned with the
// invocation result.
func (c *Context) Log(args ...any) {
	if c.checkActive() != nil {
		return
	}
	c.logMu.Lock()
	defer c.logMu.Unlock()
	line := fmt.Sprintln(args...)
	if left := maxHarnessLogs - c.logBytes; left > 0 {
		if len(line) > left {
			line = line[:left]
		}
		c.logs = append(c.logs, line)
		c.logBytes += len(line)
	}
}

// StreamStart begins an incremental HTTP response. StreamWrite starts a
// default application/octet-stream response automatically when needed.
func (c *Context) StreamStart(statusCode int, headers map[string]string) error {
	c.streamMu.Lock()
	defer c.streamMu.Unlock()
	return c.streamStartLocked(statusCode, headers)
}

func (c *Context) checkActive() error {
	select {
	case <-c.done:
		return fmt.Errorf("invocation already completed")
	default:
		return nil
	}
}
func (c *Context) streamStartLocked(statusCode int, headers map[string]string) error {
	if err := c.checkActive(); err != nil {
		return err
	}
	if c.streamStarted {
		return nil
	}
	if statusCode < 100 || statusCode > 999 {
		statusCode = 200
	}
	clean := map[string]string{}
	for key, value := range headers {
		clean[key] = value
	}
	if err := c.h.writeFrame(map[string]any{
		"type": "stream_start", "id": c.invocationID,
		"statusCode": statusCode, "headers": clean,
	}); err != nil {
		return err
	}
	c.streamStarted = true
	return nil
}

func (c *Context) StreamWrite(chunk []byte) error {
	c.streamMu.Lock()
	defer c.streamMu.Unlock()
	if !c.streamStarted {
		if err := c.streamStartLocked(200, map[string]string{"Content-Type": "application/octet-stream"}); err != nil {
			return err
		}
	}
	if err := c.checkActive(); err != nil {
		return err
	}
	for len(chunk) > 0 {
		n := len(chunk)
		if n > 65536 {
			n = 65536
		}
		if err := c.h.writeFrame(map[string]any{"type": "stream_chunk", "id": c.invocationID, "encoding": "base64", "data": base64.StdEncoding.EncodeToString(chunk[:n])}); err != nil {
			return err
		}
		chunk = chunk[n:]
	}
	return nil
}

// SSE emits one Server-Sent Event, starting the event stream on first use.
func (c *Context) SSE(event string, data any) error {
	if err := c.checkActive(); err != nil {
		return err
	}
	c.streamMu.Lock()
	defer c.streamMu.Unlock()
	if !c.streamStarted {
		if err := c.streamStartLocked(200, map[string]string{
			"Content-Type":      "text/event-stream",
			"Cache-Control":     "no-cache",
			"X-Accel-Buffering": "no",
		}); err != nil {
			return err
		}
	}
	var text string
	if value, ok := data.(string); ok {
		text = value
	} else {
		encoded, err := json.Marshal(data)
		if err != nil {
			return err
		}
		text = string(encoded)
	}
	var out strings.Builder
	if event != "" {
		out.WriteString("event: ")
		out.WriteString(strings.NewReplacer("\r", "", "\n", "").Replace(event))
		out.WriteByte('\n')
	}
	for _, line := range strings.Split(strings.NewReplacer("\r\n", "\n", "\r", "\n").Replace(text), "\n") {
		out.WriteString("data: ")
		out.WriteString(line)
		out.WriteByte('\n')
	}
	out.WriteByte('\n')
	dataBytes := []byte(out.String())
	for len(dataBytes) > 0 {
		n := len(dataBytes)
		if n > 65536 {
			n = 65536
		}
		if err := c.checkActive(); err != nil {
			return err
		}
		if err := c.h.writeFrame(map[string]any{"type": "stream_chunk", "id": c.invocationID, "encoding": "base64", "data": base64.StdEncoding.EncodeToString(dataBytes[:n])}); err != nil {
			return err
		}
		dataBytes = dataBytes[n:]
	}
	return nil
}

	if err := c.checkActive(); err != nil {

			return err

		dataBytes = dataBytes[n:]
	}
	return nil
}

type harness struct {
	conn    net.Conn
	writeMu sync.Mutex

	mu       sync.Mutex
	seq      int64
	pending  map[int64]chan callAnswer
	logs     []string
	logBytes int
}

type callAnswer struct {
	ok     bool
	result json.RawMessage
	errMsg string
}

type inFrame struct {
	Type   string          `json:"type"`
	ID     int64           `json:"id"`
	Event  json.RawMessage `json:"event"`
	CallID int64           `json:"callId"`
	OK     bool            `json:"ok"`
	Result json.RawMessage `json:"result"`
	Error  string          `json:"error"`
}

func (h *harness) writeFrame(v any) error {
	payload, err := json.Marshal(v)
	if err != nil {
		return err
	}
	if len(payload) > maxHarnessFrame {
		return fmt.Errorf("frame too large: %d", len(payload))
	}
	h.writeMu.Lock()
	defer h.writeMu.Unlock()
	var hdr [4]byte
	binary.BigEndian.PutUint32(hdr[:], uint32(len(payload)))
	if _, err := io.Copy(h.conn, bytes.NewReader(hdr[:])); err != nil {
		return err
	}
	_, err = io.Copy(h.conn, bytes.NewReader(payload))
	return err
}

func readFrame(r io.Reader) ([]byte, error) {
	var hdr [4]byte
	if _, err := io.ReadFull(r, hdr[:]); err != nil {
		return nil, err
	}
	n := binary.BigEndian.Uint32(hdr[:])
	if n > maxHarnessFrame {
		return nil, fmt.Errorf("frame too large: %d", n)
	}
	buf := make([]byte, n)
	if _, err := io.ReadFull(r, buf); err != nil {
		return nil, err
	}
	return buf, nil
}

func (h *harness) call(invocationID int64, app, tool string, input any) (json.RawMessage, error) {
	if app == "" || tool == "" {
		return nil, fmt.Errorf("context.Call: app and tool are required")
	}
	inputJSON, err := json.Marshal(input)
	if err != nil {
		return nil, fmt.Errorf("marshal call input: %w", err)
	}

	h.mu.Lock()
	h.seq++
	id := h.seq
	ch := make(chan callAnswer, 1)
	h.pending[id] = ch
	h.mu.Unlock()

	if err := h.writeFrame(map[string]any{
		"type": "call", "id": invocationID, "callId": id, "app": app, "tool": tool,
		"input": json.RawMessage(inputJSON),
	}); err != nil {
		h.mu.Lock()
		delete(h.pending, id)
		h.mu.Unlock()
		return nil, err
	}
	ans := <-ch
	if !ans.ok {
		return nil, fmt.Errorf("%s", ans.errMsg)
	}
	if len(ans.result) == 0 {
		return json.RawMessage("null"), nil
	}
	return ans.result, nil
}

func (h *harness) integration(invocationID int64, conn any, tool string, input any) (json.RawMessage, error) {
	if conn == nil {
		return nil, fmt.Errorf("context.Integration: conn (numeric id or slug) is required")
	}
	if tool == "" {
		return nil, fmt.Errorf("context.Integration: tool is required")
	}
	inputJSON, err := json.Marshal(input)
	if err != nil {
		return nil, fmt.Errorf("marshal integration input: %w", err)
	}
	// `conn` is forwarded verbatim — json.Marshal turns ints into JSON
	// numbers and strings into JSON strings, matching the sidecar's
	// number-or-string parser. Other types (bool, struct, ...) marshal
	// to something the sidecar will reject with a clear error.
	connJSON, err := json.Marshal(conn)
	if err != nil {
		return nil, fmt.Errorf("marshal integration conn: %w", err)
	}

	h.mu.Lock()
	h.seq++
	id := h.seq
	ch := make(chan callAnswer, 1)
	h.pending[id] = ch
	h.mu.Unlock()

	if err := h.writeFrame(map[string]any{
		"type": "integration", "id": invocationID, "callId": id, "tool": tool,
		"conn":  json.RawMessage(connJSON),
		"input": json.RawMessage(inputJSON),
	}); err != nil {
		h.mu.Lock()
		delete(h.pending, id)
		h.mu.Unlock()
		return nil, err
	}
	ans := <-ch
	if !ans.ok {
		return nil, fmt.Errorf("%s", ans.errMsg)
	}
	if len(ans.result) == 0 {
		return json.RawMessage("null"), nil
	}
	return ans.result, nil
}

func (h *harness) appendLog(line string) {
	h.mu.Lock()
	if h.logBytes < maxHarnessLogs {
		remaining := maxHarnessLogs - h.logBytes
		if len(line) > remaining {
			line = line[:remaining]
		}
		h.logs = append(h.logs, line)
		h.logBytes += len(line)
	}
	h.mu.Unlock()
}

func (h *harness) takeLogs() []string {
	h.mu.Lock()
	defer h.mu.Unlock()
	logs := h.logs
	h.logs = nil
	h.logBytes = 0
	return logs
}

func main() {
	file := os.NewFile(3, "apteva-fn-sock")
	conn, err := net.FileConn(file)
	_ = file.Close()
	if err != nil {
		fmt.Fprintln(os.Stderr, "harness: fd 3 is not a socket:", err)
		os.Exit(1)
	}
	h := &harness{conn: conn, pending: map[int64]chan callAnswer{}}

	if err := h.writeFrame(map[string]any{"type": "ready", "ok": true}); err != nil {
		os.Exit(1)
	}

	env := map[string]string{}
	for _, kv := range os.Environ() {
		for i := 0; i < len(kv); i++ {
			if kv[i] == '=' {
				env[kv[:i]] = kv[i+1:]
				break
			}
		}
	}
	baseCtx := &Context{
		FunctionName: os.Getenv("APTEVA_FUNCTION_NAME"),
		FunctionID:   os.Getenv("APTEVA_FUNCTION_ID"),
		Runtime:      os.Getenv("APTEVA_FUNCTION_RUNTIME"),
		Env:          env,
		h:            h,
	}

	for {
		raw, err := readFrame(conn)
		if err != nil {
			os.Exit(0) // sidecar closed the socket
		}
		var msg inFrame
		if err := json.Unmarshal(raw, &msg); err != nil {
			continue
		}
		if msg.Type == "call_result" {
			h.mu.Lock()
			ch := h.pending[msg.CallID]
			delete(h.pending, msg.CallID)
			h.mu.Unlock()
			if ch != nil {
				ch <- callAnswer{ok: msg.OK, result: msg.Result, errMsg: msg.Error}
			}
			continue
		}
		// Invocation request — run in a goroutine so the read loop
		// stays free to route call_result frames back to a Handle
		// that's blocked mid-context.Call.
		go runInvocation(h, baseCtx, msg.ID, msg.Event)
	}
}

func runInvocation(h *harness, ctx *Context, id int64, event json.RawMessage) {
	env := make(map[string]string, len(ctx.Env))
	for k, v := range ctx.Env {
		env[k] = v
	}
	invocationCtx := &Context{
		FunctionName: ctx.FunctionName,
		FunctionID:   ctx.FunctionID,
		Runtime:      ctx.Runtime,
		Env:          env,
		h:            h,
		invocationID: id,
		done:         make(chan struct{}),
	}
	var result any
	var handlerErr error
	func() {
		defer func() {
			if r := recover(); r != nil {
				handlerErr = fmt.Errorf("handler panic: %v\n%s", r, debug.Stack())
			}
		}()
		result, handlerErr = Handle(event, invocationCtx)
	}()

	close(invocationCtx.done)
	invocationCtx.logMu.Lock()
	defer invocationCtx.logMu.Unlock()
	frame := map[string]any{"type": "result", "id": id, "logs": invocationCtx.logs}
	if handlerErr != nil {
		frame["ok"] = false
		frame["error"] = handlerErr.Error()
	} else {
		frame["ok"] = true
		frame["result"] = result
	}
	if err := h.writeFrame(frame); err != nil {
		if err = h.writeFrame(map[string]any{"type": "result", "id": id, "ok": false, "error": "Response serialization failed: " + err.Error()}); err != nil {
			os.Exit(1)
		}
	}
}
